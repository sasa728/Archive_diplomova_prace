# --- WAZUH - VLASTNÍ PRAVIDLA (local_rules.xml) ---
# Přidána podpora pro MITRE ATT&CK tagování (T1110 - Brute Force)
<group name="local,syslog,sshd,">
  <rule id="5760" level="10" overwrite="yes">
    <if_sid>5700,5716</if_sid>
    <match>Failed password|Failed keyboard|authentication error</match>
    <description>sshd: authentication failed (Medium Severity)</description>
    <group>authentication_failed,soc_medium</group>
    <mitre>
      <id>T1110</id>
    </mitre>
  </rule>
  <rule id="5710" level="15" overwrite="yes">
    <if_sid>5700</if_sid>
    <match>illegal user|invalid user</match>
    <description>sshd: Attempt to login using a NON-EXISTENT user (Critical Severity)</description>
    <group>authentication_failed,invalid_login,soc_critical</group>
    <mitre>
      <id>T1110</id>
    </mitre>
  </rule>
</group>

== GRAFANA: CTI DASHBOARDS ===
==============================

# 1. Panel: Vulnerability Intelligence Feed (CVE)
SELECT timestamp AS "Čas skenu", ip_address AS "Zasažený stroj", nvt_name AS "Zranitelnost", cve AS "CVE ID", cvss AS "Závažnost (CVSS)", threat_level AS "Level" FROM greenbone_vulns WHERE cvss >= 5.0 ORDER BY cvss DESC;

# 2. Panel: Cyber Threat Intelligence (MITRE & AbuseIPDB)
SELECT timestamp AS "Čas detekce", src_ip AS "IP Útočníka", rule_description AS "Popis útoku", mitre_technique AS "MITRE ID", mitre_tactic AS "Taktika" FROM wazuh_alerts WHERE mitre_technique != 'N/A' OR rule_description ILIKE '%AbuseIPDB%' ORDER BY timestamp DESC;

# 1. Kompletní přehled hostů včetně OS Info ze Zabbixu:
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT ip_address, status, SUBSTRING(os_info, 1, 50) AS os, cpu_usage, ram_usage FROM zabbix_current;"

# 2. Přehled hostů detekovaných Wazuhem:
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT agent_id, hostname, os_name, status FROM hosts;"

# 3. Výpis TOP 10 zranitelností (včetně CVE):
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT ip_address, cvss, cve, SUBSTRING(nvt_name, 1, 60) FROM greenbone_vulns ORDER BY cvss DESC LIMIT 10;"

# 4. Poslední kritické security alerty (MITRE):
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT timestamp, src_ip, mitre_technique, rule_description FROM wazuh_alerts WHERE alert_level >= 10 ORDER BY timestamp DESC LIMIT 5;"
