# --- WAZUH - VLASTNÍ PRAVIDLA (local_rules.xml) ---
# V kontejneru manageru: /var/ossec/etc/rules/local_rules.xml
<group name="local,syslog,sshd,">
  <rule id="5760" level="10" overwrite="yes">
    <if_sid>5700,5716</if_sid>
    <match>Failed password|Failed keyboard|authentication error</match>
    <description>sshd: authentication failed (Medium Severity)</description>
    <group>authentication_failed,soc_medium</group>
  </rule>
  <rule id="5710" level="15" overwrite="yes">
    <if_sid>5700</if_sid>
    <match>illegal user|invalid user</match>
    <description>sshd: Attempt to login using a NON-EXISTENT user (Critical Severity)</description>
    <group>authentication_failed,invalid_login,soc_critical</group>
  </rule>
</group>

# --- WAZUH - CENTRÁLNÍ KONFIGURACE (Management -> Groups -> Agent.conf) ---

    <log_format>syslog</log_format>

</agent_config>

# Tyto příkazy slouží pro rychlé ověření, že integrátor (Python skript) správně zpracovává a ukládá data z API do Postgresu.

# 1. Kontrola aktivních hostů a OS (Základní identifikace napříč systémy):
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT agent_id, ip_address, hostname, os_name, status FROM hosts;"

# 2. Wazuh - Výpis posledních 5 bezpečnostních incidentů (Zdroj -> Cíl):
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT timestamp, src_ip, dst_ip, alert_level, SUBSTRING(rule_description, 1, 50) AS rule_desc FROM wazuh_alerts ORDER BY timestamp DESC LIMIT 5;"

# 3. Wazuh - Statistika závažnosti incidentů v databázi:
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT alert_level, COUNT(*) FROM wazuh_alerts GROUP BY alert_level ORDER BY alert_level DESC;"

# 4. Zabbix Current - Semafor aktuálního stavu a využití prostředků (UP/DOWN):
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT ip_address, status, cpu_usage, ram_usage, disk_usage, uptime_sec FROM zabbix_current;"

# 5. Zabbix History - Kontrola zápisu časové řady (Time-series data pro grafy):
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT timestamp, ip_address, status, cpu_usage, ram_usage FROM zabbix_history ORDER BY timestamp DESC LIMIT 6;"

# 6. Greenbone - Výpis TOP 15 nejzávažnějších zranitelností z posledního skenu:
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT ip_address, threat_level, cvss, port, SUBSTRING(nvt_name, 1, 50) AS nvt_short FROM greenbone_vulns ORDER BY cvss DESC LIMIT 15;"
