# 1. Příprava systému a instalace Dockeru
sudo apt update && sudo apt upgrade -y
sudo apt install -y ca-certificates curl gnupg git
sudo install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
sudo chmod a+r /etc/apt/keyrings/docker.gpg
echo "deb [arch="$(dpkg --print-architecture)" signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu "$(. /etc/os-release && echo "$VERSION_CODENAME")" stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
sudo apt update
sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin
sudo usermod -aG docker $USER
newgrp docker

# 2. Vytvoření síťové infrastruktury a složek
docker network create soc-network
cd ~
mkdir -p security/{wazuh,zabbix,greenbone,grafana,python}

docker compose up -d

# --- WAZUH - VLASTNÍ PRAVIDLA (local_rules.xml) ---
# V kontejneru manageru: /var/ossec/etc/rules/local_rules.xml
<group name="local,syslog,sshd,">
  <rule id="5760" level="10" overwrite="yes">
    <if_sid>5700,5716</if_sid>
    <match>Failed password|Failed keyboard|authentication error</match>
    <description>sshd: authentication failed (Medium Severity)</description>
    <group>authentication_failed,soc_medium</group>
  </rule>
  <rule id="5710" level="15" overwrite="yes">
    <if_sid>5700</if_sid>
    <match>illegal user|invalid user</match>
    <description>sshd: Attempt to login using a NON-EXISTENT user (Critical Severity)</description>
    <group>authentication_failed,invalid_login,soc_critical</group>
  </rule>
</group>

# --- WAZUH - CENTRÁLNÍ KONFIGURACE (Management -> Groups -> Agent.conf) ---
<agent_config>
  <localfile>
    <log_format>syslog</log_format>
    <location>/var/log/auth.log</location>
  </localfile>
</agent_config>

# --- GREENBONE ---
cd ~/security/greenbone
# (Zde byl stažen oficiální compose.yaml z dokumentace.

# Na konec souboru byla přidána služba gmp-proxy pro zpřístupnění API na portu 9390.)

== DATABÁZE: SOC DIAGNOSTIKA ==
==============================

# 1. Kompletní přehled hostů včetně OS Info ze Zabbixu:
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT ip_address, status, SUBSTRING(os_info, 1, 50) AS os, cpu_usage, ram_usage FROM zabbix_current;"

# 2. Přehled hostů detekovaných Wazuhem:
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT agent_id, hostname, os_name, status FROM hosts;"

# 3. Výpis TOP 10 zranitelností (včetně nového Windows hosta):
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT ip_address, cvss, port, SUBSTRING(nvt_name, 1, 60) FROM greenbone_vulns ORDER BY cvss DESC LIMIT 10;"

# 4. Poslední kritické security alerty (Level 10+):
docker compose exec postgres-db psql -U soc_user -d soc_data -c "SELECT timestamp, src_ip, dst_ip, alert_level, rule_description FROM wazuh_alerts WHERE alert_level >= 10 ORDER BY timestamp DESC LIMIT 5;"
